using UnityEngine;

/// <summary>
/// Oyuncu hareketi (WASD / ok tuşları). Diyalog sırasında bileşen kapatılınca karakter durur.
/// </summary>
[RequireComponent(typeof(Rigidbody2D))]
public class PlayerController : MonoBehaviour
{
    [SerializeField] private float moveSpeed = 5f;

    private Rigidbody2D _rb;
    private Vector2 _input;

    private void Awake() => _rb = GetComponent<Rigidbody2D>();

    private void Update()
    {
        _input = new Vector2(Input.GetAxisRaw("Horizontal"),
                             Input.GetAxisRaw("Vertical")).normalized;
    }

    private void FixedUpdate()
    {
        _rb.linearVelocity = _input * moveSpeed;
    }

    private void OnDisable()
    {
        _input = Vector2.zero;
        if (_rb != null) _rb.linearVelocity = Vector2.zero; // kayarak gitmesin
    }
}

using UnityEngine;

/// <summary>
/// Ortografik kamerayı, ekran oranı ne olursa olsun (16:9, 16:10, 4:3...) haritanın
/// tamamını gösterecek şekilde ayarlar. Pencere boyutu değişirse yeniden hesaplar.
/// </summary>
[RequireComponent(typeof(Camera))]
public class CameraFit : MonoBehaviour
{
    [Tooltip("Haritanın dünya birimi cinsinden boyutu (genişlik, yükseklik).")]
    [SerializeField] private Vector2 levelSize = new Vector2(32f, 16f);
    [Tooltip("Haritanın merkezi.")]
    [SerializeField] private Vector2 levelCenter = new Vector2(16f, 8f);
    [SerializeField] private float padding = 0.3f;

    private Camera _camera;
    private float _lastAspect;

    private void Awake()
    {
        _camera = GetComponent<Camera>();
        transform.position = new Vector3(levelCenter.x, levelCenter.y, transform.position.z);
        Fit();
    }

    private void Update()
    {
        if (!Mathf.Approximately(_camera.aspect, _lastAspect)) Fit();
    }

    private void Fit()
    {
        _lastAspect = _camera.aspect;
        float sizeForHeight = levelSize.y / 2f;
        float sizeForWidth = levelSize.x / 2f / _camera.aspect;
        _camera.orthographicSize = Mathf.Max(sizeForHeight, sizeForWidth) + padding;
    }
}

using UnityEngine;
using UnityEngine.SceneManagement;

/// <summary>
/// Oyun akışının sonu: bitiş ekranını gösterir. R: yeniden başlat, Esc: çık.
/// </summary>
public class GameManager : MonoBehaviour
{
    public bool IsGameOver { get; private set; }
    private string _message;

    public void EndGame(string message)
    {
        if (IsGameOver) return;
        IsGameOver = true;
        _message = message;
        Time.timeScale = 0f;
        Debug.Log("[Game] Oyun bitti: " + message);
    }

    private void Update()
    {
        if (!IsGameOver) return;
        if (Input.GetKeyDown(KeyCode.R))
        {
            Time.timeScale = 1f;
            SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex);
        }
        else if (Input.GetKeyDown(KeyCode.Escape))
        {
            Application.Quit();
        }
    }

    private void OnGUI()
    {
        if (!IsGameOver) return;
        var style = new GUIStyle(GUI.skin.box) { fontSize = 26, alignment = TextAnchor.MiddleCenter, wordWrap = true, richText = true };
        var rect = new Rect(Screen.width / 2f - 260, Screen.height / 2f - 90, 520, 180);
        GUI.Box(rect, GUIContent.none);
        GUI.Box(rect, $"{_message}\n\n<size=16>[R] Yeniden başla   [Esc] Çık</size>", style);
    }
}

// DOSYA: Assets/Scripts/Companion/CompanionAgent.cs

using UnityEngine;
using UnityEngine.AI;

/// <summary>
/// Companion'ın ana bileşeni. NavMeshAgent, sensör ve durum makinesini bir araya getirir.
/// Davranış kararları durum sınıflarındadır; bu sınıf onlara ortak yardımcıları sunar.
/// </summary>
[RequireComponent(typeof(NavMeshAgent))]
[RequireComponent(typeof(RaycastSensor))]
public class CompanionAgent : MonoBehaviour
{
    [Header("Referanslar")]
    [SerializeField] private Transform player;

    [Header("Takip")]
    [SerializeField] private float followStartDistance = 2.5f;
    [SerializeField] private float followStopDistance = 1.5f;

    [Header("Görev")]
    [SerializeField] private float arriveThreshold = 0.15f;
    [SerializeField] private float interactDuration = 0.6f;
    [SerializeField] private float stuckTimeout = 2f;
    [Tooltip("Yol kapalıyken (örn. kapı kapalı) en fazla kaç saniye beklenir.")]
    [SerializeField] private float waitForPathTimeout = 20f;

    public NavMeshAgent Agent { get; private set; }
    public RaycastSensor Sensor { get; private set; }
    public CompanionStateMachine StateMachine { get; private set; }
    public IInteractable CurrentTarget { get; private set; }
    public IHintProvider HintProvider { get; private set; }

    public Transform Player => player;
    public float FollowStartDistance => followStartDistance;
    public float FollowStopDistance => followStopDistance;
    public float InteractDuration => interactDuration;
    public float StuckTimeout => stuckTimeout;
    public float WaitForPathTimeout => waitForPathTimeout;

    // Durumlar bir kez oluşturulur ve tekrar kullanılır.
    public IdleState Idle { get; private set; }
    public FollowState Follow { get; private set; }
    public GoToTargetState GoToTarget { get; private set; }
    public InteractState Interact { get; private set; }
    public WaitForPathState WaitForPath { get; private set; }
    public GuideState Guide { get; private set; }

    private void Awake()
    {
        Agent = GetComponent<NavMeshAgent>();
        Agent.updateRotation = false; // 2D: sprite dönmesin
        Agent.updateUpAxis = false;   // 2D: XY düzleminde kalsın

        Sensor = GetComponent<RaycastSensor>();

        StateMachine = new CompanionStateMachine();
        StateMachine.StateChanged += (previous, next) =>
            Debug.Log($"[Companion] Durum: {previous?.Name ?? "-"} -> {next.Name}");

        Idle = new IdleState(this);
        Follow = new FollowState(this);
        GoToTarget = new GoToTargetState(this);
        Interact = new InteractState(this);
        WaitForPath = new WaitForPathState(this);
        Guide = new GuideState(this);
    }

    private void Start()
    {
        if (player == null)
        {
            Debug.LogError("[Companion] Player referansı atanmadı! Inspector'dan Player'ı sürükleyin.");
            enabled = false;
            return;
        }

        SnapToNavMesh();
        StateMachine.ChangeState(Idle);
    }

    private void Update()
    {
        Sensor.Scan(Agent.velocity);
        StateMachine.Tick();
    }

    // ---------- Dışarıdan gelen komutlar ----------

    /// <summary>Companion'a bir nesneyle etkileşim görevi verir. Kabul edilirse true döner.</summary>
    public bool TryCommandInteract(IInteractable target)
    {
        if (target == null || !target.CanInteract) return false;

        // Zaten bir hedefe gidiyorsa / bekliyorsa, yeni hedef için yolu baştan planlasın.
        if (StateMachine.Current == GoToTarget || StateMachine.Current == WaitForPath)
            StateMachine.ChangeState(Idle);

        CurrentTarget = target;
        StateMachine.ChangeState(GoToTarget);
        // Yol kapalıysa WaitForPath'te bekler; o da kabul edilmiş bir görevdir.
        return StateMachine.Current == GoToTarget || StateMachine.Current == WaitForPath;
    }

    /// <summary>Companion'a bir ipucu kaynağını (bulmaca) gösterme görevi verir.</summary>
    public bool TryCommandGuide(IHintProvider provider)
    {
        if (provider == null || !provider.HasHint) return false;

        ClearTarget();
        if (StateMachine.Current == Guide) StateMachine.ChangeState(Idle);
        HintProvider = provider;
        StateMachine.ChangeState(Guide);
        return true;
    }

    public void CommandFollow()
    {
        ClearTarget();
        ClearHint();
        StateMachine.ChangeState(Follow);
    }

    // ---------- Durumların kullandığı yardımcılar ----------

    public void MoveTo(Vector3 position)
    {
        Agent.isStopped = false;
        Agent.SetDestination(position);
    }

    public void StopMoving()
    {
        if (Agent.isOnNavMesh) Agent.ResetPath();
    }

    public bool HasArrived()
    {
        if (Agent.pathPending) return false;
        return Vector2.Distance(transform.position, Agent.destination) <= Agent.stoppingDistance + arriveThreshold;
    }

    /// <summary>
    /// Mevcut hedefe (CurrentTarget) TAM bir NavMesh yolu var mı? Ajanın areaMask'i dikkate alınır,
    /// yani kapalı kapıların alanları yol hesabına girmez.
    /// </summary>
    public bool TryGetPathToTarget(out NavMeshPath path)
    {
        path = null;
        if (CurrentTarget == null || !Agent.isOnNavMesh) return false;

        Vector3 point = CurrentTarget.InteractionPoint.position;
        // Hedef nokta NavMesh'in biraz dışında olabilir (duvar dibi): en yakın yürünebilir noktayı bul.
        if (!NavMesh.SamplePosition(point, out NavMeshHit hit, 1f, NavMesh.AllAreas)) return false;

        path = new NavMeshPath();
        return Agent.CalculatePath(hit.position, path) && path.status == NavMeshPathStatus.PathComplete;
    }

    public float DistanceToPlayer() => Vector2.Distance(transform.position, player.position);

    public void ClearTarget() => CurrentTarget = null;
    public void ClearHint() => HintProvider = null;

    private void SnapToNavMesh()
    {
        if (NavMesh.SamplePosition(transform.position, out NavMeshHit hit, 2f, NavMesh.AllAreas))
            Agent.Warp(hit.position);
        else
            Debug.LogError("[Companion] Başlangıç noktası NavMesh üzerinde değil! NavMesh'i bake ettiniz mi?");
    }
}

using System;

/// <summary>
/// Durumlar arası geçişi yöneten sade C# sınıfı (MonoBehaviour değil).
/// Geçiş kuralları durumların içindedir; bu sınıf sadece geçişi uygular.
/// </summary>
public class CompanionStateMachine
{
    public ICompanionState Current { get; private set; }

    /// <summary>(önceki, yeni) — loglama ve UI için.</summary>
    public event Action<ICompanionState, ICompanionState> StateChanged;

    public void ChangeState(ICompanionState next)
    {
        if (next == null || next == Current) return;

        ICompanionState previous = Current;
        previous?.Exit();
        Current = next;
        StateChanged?.Invoke(previous, next);
        Current.Enter();
    }

    public void Tick() => Current?.Tick();
}

using UnityEngine;

/// <summary>
/// Companion'ın "gözleri": hareket yönünde yelpaze şeklinde Physics2D ışınları atar.
/// - ObstacleAhead: önde engel var mı (takılma tespitinde kullanılır)
/// - HasLineOfSight: iki nokta arasında duvar var mı (etkileşim doğrulamasında kullanılır)
/// Scene görünümünde ışınlar yeşil (boş) / kırmızı (engel) çizilir.
/// </summary>
public class RaycastSensor : MonoBehaviour
{
    [SerializeField] private LayerMask obstacleMask;
    [SerializeField] private float range = 1.2f;
    [SerializeField, Range(1, 9)] private int rayCount = 5;
    [SerializeField, Range(0f, 180f)] private float spreadAngle = 70f;

    public bool ObstacleAhead { get; private set; }
    public float NearestObstacleDistance { get; private set; } = float.PositiveInfinity;

    private Vector2 _forward = Vector2.right;

    /// <summary>Verilen yönde tarama yapar. Yön sıfırsa son yön kullanılır.</summary>
    public void Scan(Vector2 direction)
    {
        if (direction.sqrMagnitude > 0.0001f) _forward = direction.normalized;

        ObstacleAhead = false;
        NearestObstacleDistance = float.PositiveInfinity;

        for (int i = 0; i < rayCount; i++)
        {
            Vector2 dir = RayDirection(i);
            RaycastHit2D hit = Physics2D.Raycast(transform.position, dir, range, obstacleMask);
            if (hit.collider != null)
            {
                ObstacleAhead = true;
                NearestObstacleDistance = Mathf.Min(NearestObstacleDistance, hit.distance);
            }
        }
    }

    public bool HasLineOfSight(Vector2 targetPosition)
    {
        return Physics2D.Linecast(transform.position, targetPosition, obstacleMask).collider == null;
    }

    private Vector2 RayDirection(int index)
    {
        float t = rayCount == 1 ? 0.5f : (float)index / (rayCount - 1);
        float angle = Mathf.Lerp(-spreadAngle / 2f, spreadAngle / 2f, t);
        return Quaternion.Euler(0f, 0f, angle) * _forward;
    }

    private void OnDrawGizmos()
    {
        for (int i = 0; i < rayCount; i++)
        {
            Vector2 dir = RayDirection(i);
            RaycastHit2D hit = Physics2D.Raycast(transform.position, dir, range, obstacleMask);
            Gizmos.color = hit.collider != null ? Color.red : Color.green;
            float length = hit.collider != null ? hit.distance : range;
            Gizmos.DrawLine(transform.position, (Vector2)transform.position + dir * length);
        }
    }
}

using UnityEngine;

/// <summary>
/// Takip: NavMesh üzerinden oyuncuya yol bulur. Yeterince yaklaşınca Idle'a döner.
/// Hedef her karede değil, belirli aralıklarla güncellenir (performans).
/// </summary>
public class FollowState : CompanionStateBase
{
    private const float RepathInterval = 0.25f;
    private float _repathTimer;

    public FollowState(CompanionAgent companion) : base(companion) { }

    public override string Name => "Follow";

    public override void Enter()
    {
        _repathTimer = 0f;
        Companion.MoveTo(Companion.Player.position);
    }

    public override void Tick()
    {
        if (Companion.DistanceToPlayer() <= Companion.FollowStopDistance)
        {
            Companion.StateMachine.ChangeState(Companion.Idle);
            return;
        }

        _repathTimer += Time.deltaTime;
        if (_repathTimer >= RepathInterval)
        {
            _repathTimer = 0f;
            Companion.MoveTo(Companion.Player.position);
        }
    }
}

// DOSYA: Assets/Scripts/Companion/States/GoToTargetState.cs
// ---------------------------------------------------------------------
using UnityEngine;
using UnityEngine.AI;

/// <summary>
/// Hedefe gitme: NavMesh ile etkileşim noktasına tam (complete) bir yol hesaplar ve izler.
/// - Yol kapalıysa (örn. kapı kapalı): WaitForPath durumuna geçer.
/// - Yol sırasında kapanırsa (yol "partial" olursa): WaitForPath durumuna geçer.
/// - Takılırsa: Raycast sensörü ile sebebi teşhis eder, yolu yeniden hesaplar; sınırlı deneme.
/// - Varınca: Interact durumuna geçer.
/// </summary>
public class GoToTargetState : CompanionStateBase
{
    private const int MaxRepathAttempts = 3;

    private float _stuckTimer;
    private int _repathAttempts;

    public GoToTargetState(CompanionAgent companion) : base(companion) { }

    public override string Name => "GoToTarget";

    public override void Enter()
    {
        _stuckTimer = 0f;
        _repathAttempts = 0;

        if (Companion.CurrentTarget == null)
        {
            Abort("Hedef yok.");
            return;
        }

        if (Companion.TryGetPathToTarget(out NavMeshPath path))
            Companion.Agent.SetPath(path);
        else
            Companion.StateMachine.ChangeState(Companion.WaitForPath);
    }

    public override void Tick()
    {
        IInteractable target = Companion.CurrentTarget;
        if (target == null || !target.CanInteract)
        {
            Abort("Hedef artık geçerli değil.");
            return;
        }

        NavMeshAgent agent = Companion.Agent;

        // Yol yürürken kapandıysa (kapı kapandı): tam yol yok, bekle.
        if (!agent.pathPending && agent.pathStatus != NavMeshPathStatus.PathComplete)
        {
            Companion.StateMachine.ChangeState(Companion.WaitForPath);
            return;
        }

        if (Companion.HasArrived())
        {
            Companion.StateMachine.ChangeState(Companion.Interact);
            return;
        }

        // Takılma tespiti: yol hazır ama ajan neredeyse hiç hareket etmiyor.
        bool barelyMoving = agent.velocity.sqrMagnitude < 0.01f;
        _stuckTimer = (!agent.pathPending && barelyMoving) ? _stuckTimer + Time.deltaTime : 0f;

        if (_stuckTimer >= Companion.StuckTimeout)
        {
            _stuckTimer = 0f;
            _repathAttempts++;

            string reason = Companion.Sensor.ObstacleAhead ? "önde engel var" : "hareket yok";
            Debug.LogWarning($"[Companion] Takıldı ({reason}). Yol yeniden hesaplanıyor ({_repathAttempts}/{MaxRepathAttempts}).");

            if (_repathAttempts > MaxRepathAttempts)
                Abort("Takılma çözülemedi.");
            else if (Companion.TryGetPathToTarget(out NavMeshPath path))
                agent.SetPath(path);
            else
                Companion.StateMachine.ChangeState(Companion.WaitForPath);
        }
    }

    private void Abort(string reason)
    {
        Debug.LogWarning($"[Companion] Görev iptal: {reason} Oyuncuyu takibe dönülüyor.");
        Companion.ClearTarget();
        Companion.StateMachine.ChangeState(Companion.Follow);
    }
}

using UnityEngine;

/// <summary>
/// Rehberlik: companion bir IHintProvider'ın gösterdiği konuma (örn. sıradaki doğru plaka) gider
/// ve orada bekleyerek oyuncuya işaret eder. İpucu değişince (oyuncu doğru/yanlış bastı)
/// yeni konuma NavMesh ile yol bulur. İpucu kalmayınca (bulmaca çözüldü) Follow'a döner.
/// </summary>
public class GuideState : CompanionStateBase
{
    private const float RecheckInterval = 0.5f;

    private IHintProvider _provider;
    private Vector3 _lastHint;
    private float _recheckTimer;

    public GuideState(CompanionAgent companion) : base(companion) { }

    public override string Name => "Guide";

    public override void Enter()
    {
        _provider = Companion.HintProvider;
        if (_provider == null) return;

        _provider.HintChanged += OnHintChanged;
        GoToHint();
    }

    public override void Exit()
    {
        if (_provider != null) _provider.HintChanged -= OnHintChanged;
        _provider = null;
    }

    public override void Tick()
    {
        if (_provider == null || !_provider.HasHint)
        {
            Companion.ClearHint();
            Companion.StateMachine.ChangeState(Companion.Follow);
            return;
        }

        // Güvenlik ağı: olay kaçırılırsa da hedef güncel kalsın.
        _recheckTimer += Time.deltaTime;
        if (_recheckTimer >= RecheckInterval)
        {
            _recheckTimer = 0f;
            if (Vector2.Distance(_provider.HintPosition, _lastHint) > 0.1f) GoToHint();
        }
    }

    private void OnHintChanged()
    {
        if (_provider != null && _provider.HasHint) GoToHint();
    }

    private void GoToHint()
    {
        _lastHint = _provider.HintPosition;
        Companion.MoveTo(_lastHint);
        Debug.Log("[Companion] Sıradaki doğru plakayı gösteriyor.");
    }
}

/// <summary>
/// Etkileşim: hedefe varınca durur, kısa bir süre "çalışır", Raycast ile hedefi
/// gerçekten gördüğünü doğrular ve etkileşimi tetikler. Sonra takibe döner.

{

    public InteractState(CompanionAgent companion) : base(companion) { }

    public override string Name => "Interact";

    public override void Enter()
    {
        _timer = 0f;
        Companion.StopMoving();
    }

    public override void Tick()
    {
        _timer += Time.deltaTime;
        if (_timer < Companion.InteractDuration) return;

        IInteractable target = Companion.CurrentTarget;

        if (target != null && target.CanInteract)
        {
            Vector2 targetPos = target.InteractionPoint.position;
            if (Companion.Sensor.HasLineOfSight(targetPos))

            }
            else
            {
                Debug.LogWarning("[Companion] Hedef görüş hattında değil, etkileşim yapılamadı.");
            }
        }

        Companion.ClearTarget();
        Companion.StateMachine.ChangeState(Companion.Follow);
    }
}

using UnityEngine;
using UnityEngine.AI;

/// <summary>
/// Yol bekleme: hedefe giden yol şu an kapalı (örn. tünel kapısı kapalı).
/// Companion hedefe ulaşabileceği en yakın noktaya gider ve düzenli aralıklarla yolu yeniden dener.
/// - Yol açılırsa: GoToTarget'a döner.
/// - Süre dolarsa: görevi bırakır, Follow'a döner.
/// </summary>
public class WaitForPathState : CompanionStateBase
{
    private const float CheckInterval = 0.25f;

    private float _waitTimer;
    private float _checkTimer;

    public WaitForPathState(CompanionAgent companion) : base(companion) { }

    public override string Name => "WaitForPath";

    public override void Enter()
    {
        _waitTimer = 0f;
        _checkTimer = 0f;
        Debug.Log("[Companion] Hedefe giden yol kapalı. En yakın noktada yolun açılması bekleniyor.");

        // Kısmi yol: hedefe ulaşılabilecek en yakın noktaya git (örn. kapalı kapının önü).
        if (Companion.CurrentTarget != null)
            Companion.MoveTo(Companion.CurrentTarget.InteractionPoint.position);
    }

    public override void Tick()
    {
        IInteractable target = Companion.CurrentTarget;
        if (target == null || !target.CanInteract)
        {
            Companion.ClearTarget();
            Companion.StateMachine.ChangeState(Companion.Follow);
            return;
        }

        _waitTimer += Time.deltaTime;
        if (_waitTimer >= Companion.WaitForPathTimeout)
        {
            Debug.LogWarning($"[Companion] {Companion.WaitForPathTimeout:0} sn boyunca yol açılmadı. Görev bırakıldı.");
            Companion.ClearTarget();
            Companion.StateMachine.ChangeState(Companion.Follow);
            return;
        }

        _checkTimer += Time.deltaTime;
        if (_checkTimer < CheckInterval) return;
        _checkTimer = 0f;

        if (Companion.TryGetPathToTarget(out NavMeshPath _))
        {
            Debug.Log("[Companion] Yol açıldı, hedefe devam ediliyor.");
            Companion.StateMachine.ChangeState(Companion.GoToTarget);
        }
    }
}

using System;
using UnityEngine;
using UnityEngine.AI;

/// <summary>
/// Kapı: kapalıyken oyuncuyu fiziksel olarak engeller (collider), açılınca geçilebilir olur.
/// NavMesh tarafında kapının kendi alanı (Navigation Area) vardır: kapı kapalıyken bu alan
/// ajanların areaMask'inden çıkarılır, açılınca eklenir. Böylece companion kapalı kapıdan
/// yol çizmez, kapı açılınca yolu otomatik olarak yeniden hesaplar.
/// Şalter, basınç plakası veya NPC (door_open: true) tarafından açılabilir.
/// </summary>
[RequireComponent(typeof(Collider2D))]
[RequireComponent(typeof(SpriteRenderer))]
public class Door : MonoBehaviour
{
    [SerializeField, Range(0f, 1f)] private float openAlpha = 0.15f;

    [Tooltip("Bu kapının Navigation Modifier'ında seçili alan adı (örn. TunnelGate). Boşsa NavMesh'e dokunulmaz.")]
    [SerializeField] private string navMeshArea = "";

    private Collider2D _collider;
    private SpriteRenderer _renderer;
    private Color _closedColor;
    private int _areaBit;

    public bool IsOpen { get; private set; }
    public event Action Opened;
    public event Action Closed;

    private void Awake()
    {
        _collider = GetComponent<Collider2D>();
        _renderer = GetComponent<SpriteRenderer>();
        _closedColor = _renderer.color;

        if (!string.IsNullOrEmpty(navMeshArea))
        {
            int area = NavMesh.GetAreaFromName(navMeshArea);
            if (area < 0)
                Debug.LogWarning($"[Door] {name}: '{navMeshArea}' adında bir NavMesh alanı yok (Navigation → Areas).");
            else
                _areaBit = 1 << area;
        }
    }

    private void Start() => ApplyNavMeshState();

    public void Open()
    {
        if (IsOpen) return;

        IsOpen = true;
        _collider.enabled = false;

        Color c = _closedColor;
        c.a = openAlpha;
        _renderer.color = c;

        ApplyNavMeshState();
        Debug.Log($"[Door] {name} açıldı.");
        Opened?.Invoke();
    }

    public void Close()
    {
        if (!IsOpen) return;

        IsOpen = false;
        _collider.enabled = true;
        _renderer.color = _closedColor;

        ApplyNavMeshState();
        Debug.Log($"[Door] {name} kapandı.");
        Closed?.Invoke();
    }

    /// <summary>Kapının durumunu sahnedeki tüm NavMesh ajanlarının alan maskesine yansıtır.</summary>
    private void ApplyNavMeshState()
    {
        if (_areaBit == 0) return;

        foreach (NavMeshAgent agent in FindObjectsByType<NavMeshAgent>(FindObjectsSortMode.None))
        {
            agent.areaMask = IsOpen ? agent.areaMask | _areaBit : agent.areaMask & ~_areaBit;

            // Mevcut yolu yeni duruma göre yeniden hesaplat (kapı kapandıysa yol "partial" olur).
            if (agent.isOnNavMesh && agent.hasPath)
                agent.SetDestination(agent.destination);
        }
    }
}

using UnityEngine;

/// <summary>
/// Basınç plakası: üzerinde fiziksel bir gövde (oyuncu) durduğu sürece basılı kalır.
/// Basılınca On Pressed, bırakılınca On Released olayları çalışır
/// (örn. Door.Open / Door.Close). Bağlantı Inspector'dan yapılır.
/// </summary>
[RequireComponent(typeof(Collider2D))]
[RequireComponent(typeof(SpriteRenderer))]
public class PressurePlate : MonoBehaviour
{
    [SerializeField] private Color pressedColor = new Color(1f, 0.85f, 0.2f);
    [SerializeField] private UnityEvent onPressed;
    [SerializeField] private UnityEvent onReleased;

    private SpriteRenderer _renderer;
    private Color _idleColor;
    private int _bodiesOnPlate;

    public bool IsPressed => _bodiesOnPlate > 0;

    private void Awake()
    {
        _renderer = GetComponent<SpriteRenderer>();
        _idleColor = _renderer.color;
        GetComponent<Collider2D>().isTrigger = true;
    }

    private void OnTriggerEnter2D(Collider2D other)
    {
        if (other.attachedRigidbody == null) return; // sadece fiziksel gövdeler (oyuncu) basabilir

        _bodiesOnPlate++;
        if (_bodiesOnPlate == 1)
        {
            _renderer.color = pressedColor;
            Debug.Log($"[Plate] {name} basıldı.");
            onPressed?.Invoke();
        }
    }

    private void OnTriggerExit2D(Collider2D other)
    {
        if (other.attachedRigidbody == null) return;

        _bodiesOnPlate = Mathf.Max(0, _bodiesOnPlate - 1);
        if (_bodiesOnPlate == 0)
        {
            _renderer.color = _idleColor;
            Debug.Log($"[Plate] {name} bırakıldı.");
            onReleased?.Invoke();
        }
    }
}

using System;
using System.Collections;
using System.Linq;
using UnityEngine;

/// <summary>
/// Şifreli plaka bulmacası: her oyunda rastgele bir şifre üretilir (örn. 7-2-9-4).
/// Oyuncu plakalara doğru sırayla basarsa kapı açılır; yanlış plakada ilerleme sıfırlanır.
/// IHintProvider olarak companion'a "sıradaki doğru plakanın" konumunu verir.
/// </summary>
public class SequencePuzzle : MonoBehaviour, IHintProvider
{
    [SerializeField] private Door door;
    [SerializeField, Range(2, 6)] private int codeLength = 4;
    [SerializeField] private Color correctColor = new Color(0.3f, 0.9f, 0.3f);
    [SerializeField] private Color wrongColor = new Color(0.95f, 0.25f, 0.25f);
    [SerializeField] private Color solvedColor = new Color(1f, 0.85f, 0.2f);
    [SerializeField] private float wrongFlashSeconds = 0.6f;
    [SerializeField] private KeyCode hintKeyLabel = KeyCode.Q;

    private NumberPlate[] _plates;
    private int[] _code;
    private int _progress;
    private string _message;
    private float _messageUntil;
    private Transform _player;

    public bool IsSolved { get; private set; }
    public bool HasHint => !IsSolved && _code != null && _code.Length > 0;
    public Vector3 HintPosition => PlateFor(_code[_progress]).transform.position;
    public event Action HintChanged;

    private void Awake()
    {
        _plates = FindObjectsByType<NumberPlate>(FindObjectsSortMode.None);
        if (door == null) door = GameObject.Find("PuzzleDoor")?.GetComponent<Door>();
        _player = FindFirstObjectByType<PlayerController>()?.transform;

        if (_plates.Length == 0)
        {
            Debug.LogError("[Puzzle] Sahnede NumberPlate yok.");
            enabled = false;
            return;
        }

        foreach (NumberPlate plate in _plates) plate.Pressed += OnPlatePressed;
        GenerateCode();
    }

    private void GenerateCode()
    {
        int[] numbers = _plates.Select(p => p.Number).Distinct().ToArray();
        int length = Mathf.Min(codeLength, numbers.Length);
        _code = new int[length];

        var rng = new System.Random();
        for (int i = 0; i < length; i++)
        {
            int n;
            do { n = numbers[rng.Next(numbers.Length)]; }
            while (i > 0 && n == _code[i - 1]); // arka arkaya aynı plaka olmasın
            _code[i] = n;
        }
        _progress = 0;
        Debug.Log("[Puzzle] Bu oyunun şifresi: " + string.Join("-", _code));
    }

    private void OnPlatePressed(NumberPlate plate)
    {
        if (IsSolved) return;

        if (plate.Number == _code[_progress])
        {
            plate.SetColor(correctColor);
            _progress++;

            if (_progress >= _code.Length)
            {
                Solve();
            }
            else
            {
                ShowMessage($"Doğru! ({_progress}/{_code.Length})", 1.5f);
                HintChanged?.Invoke();
            }
        }
        else
        {
            Debug.Log($"[Puzzle] Yanlış plaka: {plate.Number} (beklenen {_code[_progress]}). Sıfırlandı.");
            _progress = 0;
            foreach (NumberPlate p in _plates) p.ResetColor();
            StartCoroutine(FlashWrong(plate));
            ShowMessage("Yanlış plaka! Baştan başla.", 2f);
            HintChanged?.Invoke();
        }
    }

    private IEnumerator FlashWrong(NumberPlate plate)
    {
        plate.SetColor(wrongColor);
        yield return new WaitForSeconds(wrongFlashSeconds);
        if (!IsSolved && _progress == 0) plate.ResetColor();
    }

    private void Solve()
    {
        IsSolved = true;
        foreach (NumberPlate p in _plates) p.SetColor(solvedColor);
        if (door != null) door.Open();
        ShowMessage("Şifre doğru! Kapı açıldı.", 3f);
        Debug.Log("[Puzzle] Bulmaca çözüldü.");
        HintChanged?.Invoke();
    }

    private NumberPlate PlateFor(int number) => _plates.First(p => p.Number == number);

    private void ShowMessage(string text, float seconds)
    {
        _message = text;
        _messageUntil = Time.time + seconds;
    }

    private void OnGUI()
    {
        var style = new GUIStyle(GUI.skin.box) { fontSize = 16, alignment = TextAnchor.MiddleCenter };

        if (Time.time < _messageUntil)
            GUI.Box(new Rect(Screen.width / 2f - 170, 40, 340, 32), _message, style);

        if (!IsSolved && _player != null && IsPlayerNearPuzzle())
        {
            string dots = string.Concat(Enumerable.Range(0, _code.Length).Select(i => i < _progress ? "● " : "○ "));
            GUI.Box(new Rect(Screen.width / 2f - 170, 78, 340, 28),
                    $"Şifre: {dots}  [{hintKeyLabel}] Companion'dan ipucu", style);
        }
    }

    private bool IsPlayerNearPuzzle()
    {
        Vector3 center = Vector3.zero;
        foreach (NumberPlate p in _plates) center += p.transform.position;
        center /= _plates.Length;
        return Vector2.Distance(_player.position, center) < 5f;
    }

    private void OnDestroy()
    {
        if (_plates == null) return;
        foreach (NumberPlate plate in _plates) if (plate != null) plate.Pressed -= OnPlatePressed;
    }
}

using UnityEngine;

/// <summary>
/// Oyuncunun companion'a komut vermesi:
///   E: en yakın etkileşilebilir nesneye (şalter) git ve kullan
///   Q: bulmacada sıradaki doğru plakayı göster (ipucu)
///   F: beni takip et
/// Companion'ın kendisi hedefi bilmez; sadece IInteractable alır.
/// </summary>
public class CompanionCommandInput : MonoBehaviour
{
    [SerializeField] private CompanionAgent companion;
    [SerializeField] private KeyCode interactKey = KeyCode.E;
    [SerializeField] private KeyCode followKey = KeyCode.F;
    [SerializeField] private KeyCode hintKey = KeyCode.Q;

    private void Update()
    {
        if (companion == null) return;

        if (Input.GetKeyDown(interactKey))
        {
            IInteractable target = FindNearestInteractable();
            if (target == null)
                Debug.Log("[Komut] Etkileşilecek bir şey kalmadı.");
            else if (!companion.TryCommandInteract(target))
                Debug.LogWarning("[Komut] Companion hedefe ulaşamıyor.");
        }

        if (Input.GetKeyDown(hintKey))
        {
            IHintProvider provider = FindNearestHintProvider();
            if (provider == null || !companion.TryCommandGuide(provider))
                Debug.Log("[Komut] İpucu verilecek bir bulmaca yok.");
        }

        if (Input.GetKeyDown(followKey))
            companion.CommandFollow();
    }

    private IInteractable FindNearestInteractable()
    {
        IInteractable nearest = null;
        float best = float.PositiveInfinity;

        foreach (MonoBehaviour mb in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
        {
            if (mb is not IInteractable interactable || !interactable.CanInteract) continue;

            float d = Vector2.Distance(companion.transform.position, interactable.InteractionPoint.position);
            if (d < best)
            {
                best = d;
                nearest = interactable;
            }
        }
        return nearest;
    }

    private IHintProvider FindNearestHintProvider()
    {
        IHintProvider nearest = null;
        float best = float.PositiveInfinity;

        foreach (MonoBehaviour mb in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
        {
            if (mb is not IHintProvider provider || !provider.HasHint) continue;

            float d = Vector2.Distance(transform.position, provider.HintPosition);
            if (d < best)
            {
                best = d;
                nearest = provider;
            }
        }
        return nearest;
    }
}

/// <summary>LLM katmanındaki tüm hatalar tek tip olarak bu istisnayla bildirilir.</summary>
public class LLMException : Exception
{
    public LLMErrorKind Kind { get; }

    public LLMException(LLMErrorKind kind, string message, Exception inner = null)
        : base(message, inner)
    {
        Kind = kind;
    }

    /// <summary>Oyuncuya gösterilecek anlaşılır mesaj.</summary>
    public string UserMessage => Kind switch
    {
        LLMErrorKind.ServerUnreachable => "LLM sunucusuna ulaşılamıyor. Ollama çalışıyor mu? (ollama serve)",
        LLMErrorKind.Timeout => "Model zamanında yanıt vermedi. Tekrar deneyin.",
        LLMErrorKind.ModelNotFound => "Model yüklü değil. Terminalde 'ollama pull <model>' çalıştırın.",
        LLMErrorKind.Cancelled => "İstek iptal edildi.",
        _ => "Modelden geçersiz bir yanıt geldi. Tekrar deneyin."
    };
}

using System;
using System.Net.Http;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine;

/// <summary>
/// Ollama /api/chat uç noktasıyla konuşan ILLMClient uygulaması.
/// HttpClient + async/await kullanır: ağ beklemesi ana thread'i (frame rate'i) bloklamaz.
/// Tüm ağ/format hataları LLMException'a çevrilir.
/// </summary>
public class OllamaClient : ILLMClient, IDisposable
{
    private readonly OllamaConfig _config;
    private readonly HttpClient _http;

    public OllamaClient(OllamaConfig config)
    {
        _config = config;
        // Zaman aşımını kendimiz (CancellationToken ile) yönetiyoruz.
        _http = new HttpClient { Timeout = System.Threading.Timeout.InfiniteTimeSpan };
    }

    public async Task<string> CompleteAsync(LLMRequest request, CancellationToken cancellationToken)
    {
        string body = BuildRequestJson(request);

        using var timeoutCts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        timeoutCts.CancelAfter(TimeSpan.FromSeconds(_config.timeoutSeconds));

        HttpResponseMessage response;
        string responseText;
        try
        {
            using var content = new StringContent(body, Encoding.UTF8, "application/json");
            response = await _http.PostAsync(Url("/api/chat"), content, timeoutCts.Token);
            responseText = await response.Content.ReadAsStringAsync();
        }
        catch (OperationCanceledException e)
        {
            if (cancellationToken.IsCancellationRequested)
                throw new LLMException(LLMErrorKind.Cancelled, "İstek iptal edildi.", e);
            throw new LLMException(LLMErrorKind.Timeout, $"{_config.timeoutSeconds} sn içinde yanıt gelmedi.", e);
        }
        catch (HttpRequestException e)
        {
            throw new LLMException(LLMErrorKind.ServerUnreachable, $"Sunucuya bağlanılamadı: {e.Message}", e);
        }

        if (!response.IsSuccessStatusCode)
        {
            string error = TryReadError(responseText);
            bool modelMissing = (int)response.StatusCode == 404 || (error?.Contains("not found") ?? false);
            throw new LLMException(
                modelMissing ? LLMErrorKind.ModelNotFound : LLMErrorKind.BadResponse,
                $"HTTP {(int)response.StatusCode}: {error ?? responseText}");
        }

        OllamaChatResponse parsed;
        try
        {
            parsed = JsonUtility.FromJson<OllamaChatResponse>(responseText);
        }
        catch (Exception e)
        {
            throw new LLMException(LLMErrorKind.BadResponse, "Sunucu yanıtı çözümlenemedi.", e);
        }

        if (parsed?.message == null || string.IsNullOrWhiteSpace(parsed.message.content))
            throw new LLMException(LLMErrorKind.BadResponse, "Model boş yanıt döndürdü.");

        return parsed.message.content;
    }

    public async Task<bool> IsAvailableAsync(CancellationToken cancellationToken)
    {
        try
        {
            using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
            cts.CancelAfter(TimeSpan.FromSeconds(3));
            HttpResponseMessage response = await _http.GetAsync(Url("/api/tags"), cts.Token);
            if (!response.IsSuccessStatusCode) return false;

            string json = await response.Content.ReadAsStringAsync();
            OllamaTagsResponse tags = JsonUtility.FromJson<OllamaTagsResponse>(json);
            if (tags?.models == null) return false;

            foreach (OllamaModelInfo m in tags.models)
                if (m.name == _config.model || m.name == _config.model + ":latest")
                    return true;

            Debug.LogWarning($"[Ollama] Sunucu açık ama '{_config.model}' modeli yüklü değil.");
            return false;
        }
        catch (Exception e)
        {
            Debug.LogWarning($"[Ollama] Sunucu kontrolü başarısız: {e.Message}");
            return false;
        }
    }

    public void Dispose() => _http.Dispose();

    // ---------------- yardımcılar ----------------

    private string Url(string path) => _config.baseUrl.TrimEnd('/') + path;

    private string BuildRequestJson(LLMRequest request)
    {
        var dto = new OllamaChatRequest
        {
            model = _config.model,
            stream = false,
            keep_alive = "15m",
            messages = new ChatMessage[request.Messages.Count],
            options = new OllamaOptions
            {
                temperature = _config.temperature,
                num_predict = _config.maxTokens
            }
        };
        for (int i = 0; i < request.Messages.Count; i++) dto.messages[i] = request.Messages[i];

        string json = JsonUtility.ToJson(dto);

        // JsonUtility iç içe serbest JSON (şema) yazamaz; "format" alanını elle ekliyoruz.
        string format = string.IsNullOrEmpty(request.JsonSchema) ? null : request.JsonSchema;
        if (format != null)
            json = json.Substring(0, json.Length - 1) + ",\"format\":" + format + "}";

        return json;
    }

    private static string TryReadError(string body)
    {
        try { return JsonUtility.FromJson<OllamaError>(body)?.error; }
        catch { return null; }
    }

    // ---------------- Ollama JSON veri sınıfları ----------------

    [Serializable] private class OllamaChatRequest
    {
        public string model;
        public ChatMessage[] messages;
        public bool stream;
        public string keep_alive;
        public OllamaOptions options;
    }

    [Serializable] private class OllamaOptions
    {
        public float temperature;
        public int num_predict;
    }

    [Serializable] private class OllamaChatResponse
    {
        public ChatMessage message;
        public bool done;
    }

    [Serializable] private class OllamaTagsResponse { public OllamaModelInfo[] models; }
    [Serializable] private class OllamaModelInfo { public string name; }
    [Serializable] private class OllamaError { public string error; }
}

/// <summary>Bir diyalog turunun sonucu: ya geçerli bir NPC yanıtı ya da oyuncuya gösterilecek hata.</summary>
public class DialogueOutcome
{
    public bool Success { get; }
    public NPCResponse Response { get; }
    public string ErrorMessage { get; }

    private DialogueOutcome(bool success, NPCResponse response, string error)
    {
        Success = success;
        Response = response;
        ErrorMessage = error;

// DOSYA: Assets/Scripts/NPC/GuardNPC.cs
// ---------------------------------------------------------------------
using System.Threading;
using UnityEngine;

/// <summary>
/// Kapıyı koruyan NPC. Bileşenleri birbirine bağlayan "composition root":
///   OllamaClient (ILLMClient) + NPCResponseParser + NPCDialogueController + IDialogueView + Door
/// Oyuncu yakına gelince [T] ile konuşma açılır, [Esc] ile kapanır.
/// Model door_open = true döndürürse kapı açılır.
/// </summary>
[RequireComponent(typeof(Collider2D))]
public class GuardNPC : MonoBehaviour
{
    [Header("Kişilik")]
    [SerializeField] private NPCPersona persona = new NPCPersona();

    [Header("LLM")]
    [SerializeField] private OllamaConfig ollama = new OllamaConfig();

    [Header("Bağlantılar")]
    [SerializeField] private Door guardedDoor;
    [SerializeField] private DialogueUI dialogueView;
    [Tooltip("Konuşurken kapatılacak bileşenler. Boş bırakılırsa PlayerController ve CompanionCommandInput otomatik bulunur.")]
    [SerializeField] private Behaviour[] disableWhileTalking;

    [Header("Etkileşim")]
    [SerializeField] private KeyCode talkKey = KeyCode.T;

    private OllamaClient _client;
    private NPCDialogueController _dialogue;
    private IDialogueView _view;
    private CancellationTokenSource _lifetimeCts;
    private bool _playerInRange;
    private bool _talking;

    private void Awake()
    {
        _lifetimeCts = new CancellationTokenSource();
        _client = new OllamaClient(ollama);
        _dialogue = new NPCDialogueController(_client, new NPCResponseParser(), persona);
        _view = dialogueView;

        if (disableWhileTalking == null || disableWhileTalking.Length == 0)
        {
            disableWhileTalking = new Behaviour[]
            {
                FindFirstObjectByType<PlayerController>(),
                FindFirstObjectByType<CompanionCommandInput>()
            };
        }
    }

    private async void Start()
    {
        if (_view == null)
        {
            Debug.LogError("[GuardNPC] Dialogue View atanmamış!");
            enabled = false;
            return;
        }

        _view.MessageSubmitted += OnPlayerMessage;
        _view.CloseRequested += EndConversation;

        // Açılışta sunucu kontrolü: kapalıysa oyun yine devam eder, sadece uyarı gösterilir.
        bool available = await _client.IsAvailableAsync(_lifetimeCts.Token);
        if (this == null) return;
        _view.SetServerStatus(available
            ? $"LLM: {ollama.model} hazır"
            : $"LLM: bağlantı yok ({ollama.model}) — Ollama'yı başlatın");
    }

    /// <summary>Inspector'da bileşenin ⋮ menüsünden: kişiliği koddaki güncel varsayılana döndürür.</summary>
    [ContextMenu("Varsayılan kişiliği yükle")]
    private void LoadDefaultPersona()
    {
        persona = new NPCPersona();
        ollama.temperature = new OllamaConfig().temperature;
#if UNITY_EDITOR
        UnityEditor.EditorUtility.SetDirty(this);
#endif
        Debug.Log("[GuardNPC] Varsayılan kişilik ve sıcaklık yüklendi.");
    }

    private void Update()
    {
        if (_playerInRange && !_talking && Input.GetKeyDown(talkKey))
            BeginConversation();
    }

    private void BeginConversation()
    {
        _talking = true;
        SetPlayerControls(false);
        _view.Open(persona.displayName);
        _view.AppendNpcMessage(persona.displayName, persona.greeting, "neutral");
    }

    private void EndConversation()
    {
        if (!_talking) return;
        _talking = false;
        _view.Close();
        SetPlayerControls(true);
    }

    private async void OnPlayerMessage(string text)
    {
        if (_dialogue.IsBusy) return;

        _view.AppendPlayerMessage(text);
        _view.SetThinking(true);

        DialogueOutcome outcome = await _dialogue.SendAsync(text, _lifetimeCts.Token);

        if (this == null) return; // sahne kapandıysa
        _view.SetThinking(false);

        if (!outcome.Success)
        {
            _view.AppendSystemMessage(outcome.ErrorMessage);
            return;
        }

        NPCResponse r = outcome.Response;
        _view.AppendNpcMessage(persona.displayName, r.reply, r.mood);

        if (r.door_open && guardedDoor != null && !guardedDoor.IsOpen)
        {
            guardedDoor.Open();
            _view.AppendSystemMessage("Kapı açıldı! [Esc] ile konuşmayı bitirip ilerleyebilirsin.");
        }
    }

    private void SetPlayerControls(bool enabledState)
    {
        if (disableWhileTalking == null) return;
        foreach (Behaviour b in disableWhileTalking)
            if (b != null) b.enabled = enabledState;
    }

    private void OnTriggerEnter2D(Collider2D other)
    {
        if (other.GetComponent<PlayerController>() != null) _playerInRange = true;
    }

    private void OnTriggerExit2D(Collider2D other)
    {
        if (other.GetComponent<PlayerController>() != null) _playerInRange = false;
    }

    private void OnGUI()
    {
        if (!_playerInRange || _talking || Camera.main == null) return;
        Vector3 p = Camera.main.WorldToScreenPoint(transform.position + Vector3.up * 0.9f);
        var style = new GUIStyle(GUI.skin.box) { fontSize = 14, alignment = TextAnchor.MiddleCenter };
        GUI.Box(new Rect(p.x - 70, Screen.height - p.y - 14, 140, 28), $"[{talkKey}] Konuş", style);
    }

    private void OnDestroy()
    {
        if (_view != null)
        {
            _view.MessageSubmitted -= OnPlayerMessage;
            _view.CloseRequested -= EndConversation;
        }
        _lifetimeCts?.Cancel();
        _lifetimeCts?.Dispose();
        _client?.Dispose();
    }
}

using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine;

/// <summary>
/// Diyalog mantığı (Unity arayüzünden bağımsız, sade C#):
/// sohbet geçmişini tutar, modele istek atar, çıktıyı ayrıştırır, gerekirse bir kez düzeltme ister.
/// Hiçbir durumda istisna dışarı sızmaz; sonuç her zaman DialogueOutcome olarak döner.
/// </summary>
public class NPCDialogueController
{
    private const int MaxParseRetries = 1;
    private const string RetryInstruction =
        "Önceki çıktın geçerli değildi. SADECE şu şemadaki JSON nesnesini döndür: " +
        "{\"reply\": string, \"mood\": \"neutral\"|\"friendly\"|\"suspicious\"|\"angry\", \"door_open\": boolean}";

    private readonly ILLMClient _client;
    private readonly IResponseParser<NPCResponse> _parser;
    private readonly NPCPersona _persona;
    private readonly int _maxHistoryMessages;
    private readonly List<ChatMessage> _history = new List<ChatMessage>();

    public bool IsBusy { get; private set; }

    public NPCDialogueController(ILLMClient client, IResponseParser<NPCResponse> parser,
                                 NPCPersona persona, int maxHistoryMessages = 12)
    {
        _client = client;
        _parser = parser;
        _persona = persona;
        _maxHistoryMessages = maxHistoryMessages;
    }

    public async Task<DialogueOutcome> SendAsync(string playerText, CancellationToken ct)
    {
        if (IsBusy) return DialogueOutcome.Fail("NPC hâlâ düşünüyor, biraz bekleyin.");
        if (string.IsNullOrWhiteSpace(playerText)) return DialogueOutcome.Fail("Boş mesaj gönderilemez.");

        IsBusy = true;
        var userMessage = ChatMessage.User(playerText.Trim());
        _history.Add(userMessage);

        try
        {
            for (int attempt = 0; attempt <= MaxParseRetries; attempt++)
            {
                List<ChatMessage> messages = BuildMessages(retry: attempt > 0);
                var stopwatch = System.Diagnostics.Stopwatch.StartNew();

                string raw = await _client.CompleteAsync(new LLMRequest(messages, NPCResponse.JsonSchema), ct);
                Debug.Log($"[NPC] Model yanıtı ({stopwatch.ElapsedMilliseconds} ms): {raw}");

                if (_parser.TryParse(raw, out NPCResponse response, out string parseError))
                {
                    // Geçmişe temiz (normalize edilmiş) JSON'u yazıyoruz ki model formatı korusun.
                    _history.Add(ChatMessage.Assistant(JsonUtility.ToJson(response)));
                    TrimHistory();
                    return DialogueOutcome.Ok(response);
                }

                Debug.LogWarning($"[NPC] Ayrıştırma hatası (deneme {attempt + 1}): {parseError}");
            }

            _history.Remove(userMessage); // oyuncu aynı mesajı tekrar deneyebilsin
            return DialogueOutcome.Fail("Bekçi ne diyeceğini şaşırdı (geçersiz yanıt). Tekrar deneyin.");
        }
        catch (LLMException e)
        {
            Debug.LogWarning($"[NPC] LLM hatası ({e.Kind}): {e.Message}");
            _history.Remove(userMessage);
            return DialogueOutcome.Fail(e.UserMessage);
        }
        catch (System.Exception e)
        {
            // Beklenmeyen her şey: oyun çökmesin.
            Debug.LogException(e);
            _history.Remove(userMessage);
            return DialogueOutcome.Fail("Beklenmeyen bir hata oluştu. Tekrar deneyin.");
        }
        finally
        {
            IsBusy = false;
        }
    }

    private List<ChatMessage> BuildMessages(bool retry)
    {
        var list = new List<ChatMessage>(_history.Count + 2) { ChatMessage.System(_persona.systemPrompt) };
        list.AddRange(_history);
        if (retry) list.Add(ChatMessage.System(RetryInstruction));
        return list;
    }

    private void TrimHistory()
    {
        while (_history.Count > _maxHistoryMessages) _history.RemoveAt(0);
    }
}

using System;

/// <summary>
/// NPC'nin modelden beklenen yapılandırılmış yanıtı.
/// Alan adları JSON şemasıyla birebir aynıdır (JsonUtility bu isimlerle eşleştirir).
/// </summary>
[Serializable]
public class NPCResponse
{
    // Alan sırası bilinçlidir: model önce sözünü ve ruh halini üretir, kapı kararını EN SON verir.
    // Böylece karar, az önce yazdığı cevapla tutarlı olur.
    public string reply;      // NPC'nin oyuncuya söylediği söz
    public string mood;       // neutral | friendly | suspicious | angry
    public bool door_open;    // true => kapı açılır (oyun durumu değişir)

    public static readonly string[] AllowedMoods = { "neutral", "friendly", "suspicious", "angry" };

    /// <summary>Ollama'ya "format" olarak gönderilen JSON şeması.</summary>
    public const string JsonSchema =
        "{\"type\":\"object\",\"properties\":{" +
        "\"reply\":{\"type\":\"string\"}," +

        "\"door_open\":{\"type\":\"boolean\"}" +
        "},\"required\":[\"reply\",\"mood\",\"door_open\"]}";

using System;
using System.Linq;
using UnityEngine;

/// <summary>
/// Model çıktısını NPCResponse'a çevirir ve doğrular.
/// Küçük modeller bazen JSON'u ``` blokları veya açıklama metniyle sarar;
/// ilk '{' ile son '}' arası alınarak bu durum tolere edilir.
/// Bozuk/eksik çıktı istisna fırlatmaz, false döner.
/// </summary>
public class NPCResponseParser : IResponseParser<NPCResponse>
{
    public bool TryParse(string raw, out NPCResponse result, out string error)
    {
        result = null;
        error = null;

        if (string.IsNullOrWhiteSpace(raw))
        {
            error = "Boş çıktı.";
            return false;
        }

        int start = raw.IndexOf('{');
        int end = raw.LastIndexOf('}');
        if (start < 0 || end <= start)
        {
            error = "Çıktıda JSON nesnesi yok.";
            return false;
        }

        string json = raw.Substring(start, end - start + 1);

        // JsonUtility eksik alanları sessizce varsayılan bırakır; zorunlu alanları elle kontrol ediyoruz.
        if (!json.Contains("\"reply\"") || !json.Contains("\"door_open\""))
        {
            error = "Zorunlu alanlar (reply, door_open) eksik.";
            return false;
        }

        try
        {
            result = JsonUtility.FromJson<NPCResponse>(json);
        }
        catch (Exception e)
        {
            error = "Geçersiz JSON: " + e.Message;
            return false;
        }

        if (result == null || string.IsNullOrWhiteSpace(result.reply))
        {
            error = "reply alanı boş.";
            result = null;
            return false;
        }

        result.reply = result.reply.Trim();
        result.mood = NormalizeMood(result.mood);
        return true;
    }

    private static string NormalizeMood(string mood)
    {
        string m = (mood ?? "").Trim().ToLowerInvariant();
        return NPCResponse.AllowedMoods.Contains(m) ? m : "neutral";
    }
}

/// <summary>
/// IDialogueView'in IMGUI ile çizilen uygulaması (Canvas kurulumu gerektirmez).
/// Ekranın alt kısmında sohbet geçmişi, yazı kutusu ve "Düşünüyor..." göstergesi.
/// Enter: gönder, Esc: kapat.
/// </summary>
public class DialogueUI : MonoBehaviour, IDialogueView
{
    private const string InputControlName = "DialogueInput";
    private const int MaxInputLength = 300;

    public event Action<string> MessageSubmitted;
    public event Action CloseRequested;

    private readonly List<(string text, Color color)> _lines = new List<(string, Color)>();
    private string _npcName = "";
    private string _input = "";
    private string _serverStatus = "LLM: kontrol ediliyor...";
    private bool _open;
    private bool _thinking;
    private bool _focusInput;
    private Vector2 _scroll;
    private GUIStyle _lineStyle, _titleStyle, _statusStyle;

    public void Open(string npcName)
    {
        _npcName = npcName;
        _open = true;
        _focusInput = true;
    }

    public void Close()
    {
        _open = false;
        _input = "";
    }

    public void AppendPlayerMessage(string text) => AddLine($"Sen: {text}", new Color(0.6f, 0.85f, 1f));

    public void AppendNpcMessage(string npcName, string text, string mood) =>
        AddLine($"{npcName} ({MoodLabel(mood)}): {text}", MoodColor(mood));

    public void AppendSystemMessage(string text) => AddLine($"! {text}", new Color(1f, 0.8f, 0.3f));

    public void SetThinking(bool thinking)
    {
        _thinking = thinking;
        if (!thinking) _focusInput = true;
    }

    public void SetServerStatus(string status) => _serverStatus = status;

    private void AddLine(string text, Color color)
    {
        _lines.Add((text, color));
        _scroll.y = float.MaxValue; // en alta kaydır
    }

    private void OnGUI()
    {
        DrawStatus();
        if (!_open) return;

        EnsureStyles();

        float w = Mathf.Min(900f, Screen.width - 40f);
        float h = Mathf.Min(320f, Screen.height * 0.45f);
        var panel = new Rect((Screen.width - w) / 2f, Screen.height - h - 20f, w, h);
        GUI.Box(panel, GUIContent.none);
        GUI.Box(panel, GUIContent.none); // daha koyu arka plan

        GUILayout.BeginArea(new Rect(panel.x + 12, panel.y + 8, panel.width - 24, panel.height - 16));
        GUILayout.Label($"{_npcName} ile konuşma   —   Enter: gönder, Esc: kapat", _titleStyle);

        _scroll = GUILayout.BeginScrollView(_scroll, GUILayout.ExpandHeight(true));
        foreach (var (text, color) in _lines)
        {
            _lineStyle.normal.textColor = color;
            GUILayout.Label(text, _lineStyle);
        }
        if (_thinking)
        {
            _lineStyle.normal.textColor = Color.gray;
            int dots = 1 + (int)(Time.unscaledTime * 3f) % 3;
            GUILayout.Label($"{_npcName} düşünüyor" + new string('.', dots), _lineStyle);
        }
        GUILayout.EndScrollView();

        HandleKeys();

        GUILayout.BeginHorizontal();
        GUI.enabled = !_thinking;
        GUI.SetNextControlName(InputControlName);
        _input = GUILayout.TextField(_input, MaxInputLength, GUILayout.Height(28));
        if (GUILayout.Button(_thinking ? "Bekleniyor..." : "Gönder", GUILayout.Width(120), GUILayout.Height(28)))
            Submit();
        GUI.enabled = true;
        GUILayout.EndHorizontal();

        GUILayout.EndArea();

        if (_focusInput && !_thinking)
        {
            GUI.FocusControl(InputControlName);
            _focusInput = false;
        }
    }

    private void HandleKeys()
    {
        Event e = Event.current;
        if (e.type != EventType.KeyDown) return;

        if (e.keyCode == KeyCode.Return || e.keyCode == KeyCode.KeypadEnter)
        {
            Submit();
            e.Use();
        }
        else if (e.keyCode == KeyCode.Escape)
        {
            CloseRequested?.Invoke();
            e.Use();
        }
    }

    private void Submit()
    {
        if (_thinking || string.IsNullOrWhiteSpace(_input)) return;
        string text = _input.Trim();
        _input = "";
        MessageSubmitted?.Invoke(text);
    }

    private void DrawStatus()
    {
        _statusStyle ??= new GUIStyle(GUI.skin.label) { fontSize = 12 };
        GUI.Label(new Rect(10, 8, 500, 22), _serverStatus, _statusStyle);
    }

    private void EnsureStyles()
    {
        _lineStyle ??= new GUIStyle(GUI.skin.label) { fontSize = 15, wordWrap = true, richText = false };
        _titleStyle ??= new GUIStyle(GUI.skin.label) { fontSize = 13, fontStyle = FontStyle.Bold };
    }

    private static string MoodLabel(string mood) => mood switch
    {
        "friendly" => "dostça",
        "suspicious" => "şüpheli",
        "angry" => "kızgın",
        _ => "nötr"
    };

    private static Color MoodColor(string mood) => mood switch
    {
        "friendly" => new Color(0.55f, 1f, 0.55f),
        "suspicious" => new Color(1f, 0.9f, 0.4f),
        "angry" => new Color(1f, 0.45f, 0.45f),
        _ => Color.white
    };
}

// ---------------------------------------------------------------------
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using UnityEngine;
using UnityEngine.AI;

/// <summary>
/// Companion tekrarlanabilirlik testi (rapordaki "başarı oranı / ortalama süre" için).
/// Play modunda F9: companion'ı rastgele (ama seed'li, yani tekrarlanabilir) başlangıç noktalarına koyar,
/// şalteri sıfırlar, görevi verir ve süreyi ölçer.
/// Sonuçlar Console'a ve proje kökündeki TestResults/ klasörüne (CSV + Markdown tablo) yazılır.
/// </summary>
public class CompanionTestRunner : MonoBehaviour
{
    [SerializeField] private CompanionAgent companion;
    [SerializeField] private Lever lever;
    [SerializeField] private Door gateDoor;
    [Tooltip("Test sırasında açık tutulan tünel kapısı (basınç plakası yerine).")]
    [SerializeField] private Door tunnelDoor;

    [Header("Test ayarları")]
    [SerializeField] private int trialCount = 20;
    [SerializeField] private float trialTimeout = 30f;
    [SerializeField] private int randomSeed = 42;
    [Tooltip("Başlangıç noktalarının seçileceği dünya alanları (x, y, genişlik, yükseklik).")]
    [SerializeField] private Rect[] spawnAreas = { new Rect(1f, 8f, 8f, 7f) };
    [Tooltip("Açıksa şalter de her denemede mekanizma odasında rastgele bir yere taşınır.")]
    [SerializeField] private bool randomizeLever = true;
    [SerializeField] private Rect leverArea = new Rect(1.5f, 1.5f, 7f, 3f);
    [SerializeField] private KeyCode runKey = KeyCode.F9;

    private struct TrialResult
    {
        public int Index;
        public Vector2 Start;
        public Vector2 LeverPos;
        public bool Success;
        public float Seconds;
        public float PathLength;
        public string Note;
    }

    private bool _running;
    private string _status = "";

    private void Awake()
    {
        if (companion == null) companion = FindFirstObjectByType<CompanionAgent>();
        if (lever == null) lever = FindFirstObjectByType<Lever>();
        if (gateDoor == null) gateDoor = GameObject.Find("GateDoor")?.GetComponent<Door>();
        if (tunnelDoor == null) tunnelDoor = GameObject.Find("TunnelDoor")?.GetComponent<Door>();
    }

    private void Update()
    {
        if (!_running && Input.GetKeyDown(runKey))
            StartCoroutine(RunAll());
    }

    private IEnumerator RunAll()
    {
        if (companion == null || lever == null)
        {
            Debug.LogError("[Test] Companion veya Lever bulunamadı.");
            yield break;
        }

        _running = true;
        var rng = new System.Random(randomSeed);
        var results = new List<TrialResult>();
        Vector3 originalLeverPos = lever.transform.position;

        for (int i = 1; i <= trialCount; i++)
        {
            _status = $"Companion testi {i}/{trialCount}...";

            lever.ResetLever();
            if (gateDoor != null) gateDoor.Close();
            if (tunnelDoor != null) tunnelDoor.Open(); // oyuncunun plakada durmasını simüle eder

            if (randomizeLever)
                lever.transform.position = RandomNavMeshPoint(rng, new[] { leverArea });

            Vector3 start = RandomNavMeshPoint(rng, spawnAreas);
            companion.Agent.Warp(start);
            yield return null;

            var result = new TrialResult
            {
                Index = i,
                Start = start,
                LeverPos = lever.transform.position,
                PathLength = PathLength(start, lever.InteractionPoint.position, companion.Agent.areaMask)
            };

            float t0 = Time.time;
            if (!companion.TryCommandInteract(lever))
            {
                result.Note = "Yol bulunamadı";
                results.Add(result);
                continue;
            }

            while (lever.CanInteract && Time.time - t0 < trialTimeout)
            {
                var state = companion.StateMachine.Current;
                if (state != companion.GoToTarget && state != companion.Interact && state != companion.WaitForPath)
                    break; // görev iptal edildi
                yield return null;
            }

            result.Success = !lever.CanInteract;
            result.Seconds = Time.time - t0;
            result.Note = result.Success ? "OK" : (Time.time - t0 >= trialTimeout ? "Zaman aşımı" : "Görev iptal");
            results.Add(result);

            yield return new WaitForSeconds(0.3f);
        }

        lever.transform.position = originalLeverPos;
        lever.ResetLever();
        if (gateDoor != null) gateDoor.Close();
        if (tunnelDoor != null) tunnelDoor.Close();

        WriteReport(results);
        _status = "";
        _running = false;
    }

    private Vector3 RandomNavMeshPoint(System.Random rng, Rect[] areas)
    {
        for (int attempt = 0; attempt < 100; attempt++)
        {
            Rect area = areas[rng.Next(areas.Length)];
            var p = new Vector3(area.x + (float)rng.NextDouble() * area.width,
                                area.y + (float)rng.NextDouble() * area.height, 0f);
            if (NavMesh.SamplePosition(p, out NavMeshHit hit, 0.5f, NavMesh.AllAreas))
                return hit.position;
        }
        return companion.transform.position;
    }

    private static float PathLength(Vector3 from, Vector3 to, int areaMask)
    {
        var path = new NavMeshPath();
        if (!NavMesh.CalculatePath(from, to, areaMask, path) || path.corners.Length < 2) return -1f;
        float length = 0f;
        for (int i = 1; i < path.corners.Length; i++) length += Vector3.Distance(path.corners[i - 1], path.corners[i]);
        return length;
    }

    private void WriteReport(List<TrialResult> results)
    {
        var ci = CultureInfo.InvariantCulture;
        int success = 0;
        float totalTime = 0f;
        foreach (var r in results) if (r.Success) { success++; totalTime += r.Seconds; }
        float rate = 100f * success / Mathf.Max(1, results.Count);
        float avg = success > 0 ? totalTime / success : 0f;

        var csv = new StringBuilder("deneme,baslangic_x,baslangic_y,salter_x,salter_y,yol_uzunlugu,basarili,sure_sn,not\n");
        var md = new StringBuilder();
        md.AppendLine($"**Deneme:** {results.Count} | **Başarı oranı:** %{rate.ToString("0.0", ci)} | **Ortalama süre:** {avg.ToString("0.00", ci)} sn | **Seed:** {randomSeed}");
        md.AppendLine();
        md.AppendLine("| # | Companion başlangıcı (x, y) | Şalter (x, y) | Yol uzunluğu | Sonuç | Süre (sn) |");
        md.AppendLine("|---|---|---|---|---|---|");

        foreach (var r in results)
        {
            string x = r.Start.x.ToString("0.00", ci), y = r.Start.y.ToString("0.00", ci);
            string lx = r.LeverPos.x.ToString("0.00", ci), ly = r.LeverPos.y.ToString("0.00", ci);
            string len = r.PathLength.ToString("0.00", ci), sec = r.Seconds.ToString("0.00", ci);
            csv.AppendLine($"{r.Index},{x},{y},{lx},{ly},{len},{(r.Success ? 1 : 0)},{sec},{r.Note}");
            md.AppendLine($"| {r.Index} | ({x}, {y}) | ({lx}, {ly}) | {len} | {(r.Success ? "✅" : "❌ " + r.Note)} | {sec} |");
        }

        string dir = Path.GetFullPath(Path.Combine(Application.dataPath, "..", "TestResults"));
        Directory.CreateDirectory(dir);
        File.WriteAllText(Path.Combine(dir, "companion_tests.csv"), csv.ToString(), Encoding.UTF8);
        File.WriteAllText(Path.Combine(dir, "companion_tests.md"), md.ToString(), Encoding.UTF8);

        Debug.Log($"[Test] Bitti. Başarı: {success}/{results.Count} (%{rate:0.0}), ortalama süre: {avg:0.00} sn\n" +
                  $"Sonuçlar: {dir}\n\n{md}");
    }

    private void OnGUI()
    {
        if (string.IsNullOrEmpty(_status)) return;
        GUI.Box(new Rect(Screen.width - 260, 8, 250, 26), _status);
    }
}

using UnityEditor;
using UnityEngine;
using UnityEngine.Tilemaps;

/// <summary>
/// Menü: Yazlab → Görselleri Uygula
/// Assets/Art/Sprites altındaki pixel art görselleri içe aktarma ayarlarıyla (32 PPU, Point filtre)
/// hazırlar ve tile'lara / sahne objelerine atar. Renk tonlarını beyaza çeker ki görsel doğru görünsün.
/// Ctrl+Z ile sahne değişiklikleri geri alınabilir.
/// </summary>
public static class ArtApplier
{
    private const string Folder = "Assets/Art/Sprites/";

    [MenuItem("Yazlab/Görselleri Uygula")]
    public static void Apply()
    {
        string[] names = { "grass", "wall", "player", "companion", "guard", "door", "plate", "lever" };
        foreach (string n in names) ConfigureImporter(Folder + n + ".png");
        AssetDatabase.Refresh();

        // Tile'lar
        SetTile("Ground", "grass");
        SetTile("Wall", "wall");
        foreach (Tilemap map in Object.FindObjectsByType<Tilemap>(FindObjectsSortMode.None))
        {
            Undo.RecordObject(map, "Tilemap rengi");
            map.color = Color.white;
            map.RefreshAllTiles();
        }

        // Sahne objeleri
        SetSprite("Player", "player");
        SetSprite("Companion", "companion");
        SetSprite("Guard", "guard");
        SetSprite("Lever", "lever");
        SetSprite("PressurePlate", "plate");
        foreach (string door in new[] { "GateDoor", "TunnelDoor", "GuardDoor", "PuzzleDoor" })
            SetSprite(door, "door");
        for (int i = 1; i <= 9; i++)
            SetSprite("Plate" + i, "plate", keepColor: true);

        UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(
            UnityEditor.SceneManagement.EditorSceneManager.GetActiveScene());
        Debug.Log("[ArtApplier] Görseller uygulandı. Navigation → Bake yapıp sahneyi kaydedin (Ctrl+S).");
    }

    private static void ConfigureImporter(string path)
    {
        if (AssetImporter.GetAtPath(path) is not TextureImporter ti)
        {
            Debug.LogWarning("[ArtApplier] Bulunamadı: " + path);
            return;
        }
        ti.textureType = TextureImporterType.Sprite;
        ti.spriteImportMode = SpriteImportMode.Single;
        ti.spritePixelsPerUnit = 32;
        ti.filterMode = FilterMode.Point;
        ti.textureCompression = TextureImporterCompression.Uncompressed;
        ti.mipmapEnabled = false;
        ti.SaveAndReimport();
    }

    private static Sprite LoadSprite(string name) =>
        AssetDatabase.LoadAssetAtPath<Sprite>(Folder + name + ".png");

    private static void SetTile(string tileName, string spriteName)
    {
        Sprite sprite = LoadSprite(spriteName);
        foreach (string guid in AssetDatabase.FindAssets(tileName + " t:Tile"))
        {
            var tile = AssetDatabase.LoadAssetAtPath<Tile>(AssetDatabase.GUIDToAssetPath(guid));
            if (tile == null || tile.name != tileName) continue;
            Undo.RecordObject(tile, "Tile görseli");
            tile.sprite = sprite;
            tile.color = Color.white;
            EditorUtility.SetDirty(tile);
        }
    }

    private static void SetSprite(string objectName, string spriteName, bool keepColor = false)
    {
        GameObject go = GameObject.Find(objectName);
        if (go == null || !go.TryGetComponent(out SpriteRenderer sr))
        {
            Debug.LogWarning($"[ArtApplier] '{objectName}' bulunamadı, atlandı.");
            return;
        }
        Undo.RecordObject(sr, "Sprite");
        sr.sprite = LoadSprite(spriteName);
        if (!keepColor) sr.color = Color.white;
    }
}

using System.Collections.Generic;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.Tilemaps;

/// <summary>
/// Menü: Yazlab → Haritayı Oluştur
/// Aşağıdaki metin haritasını Ground/Walls tilemap'lerine çizer ve sahne objelerini yerleştirir.
///   #  duvar          .  zemin
///   P  oyuncu         C  companion        L  şalter
///   g  portcullis (GateDoor)              k  kale kapısı (GuardDoor)
///   t  tünel kapısı (TunnelDoor)          p  basınç plakası (PressurePlate)
///   q  bulmaca kapısı (PuzzleDoor)        1-9 numaralı plakalar (otomatik oluşturulur)
///   N  bekçi (Guard)  E  çıkış alanı (EndZone)
/// Ctrl+Z ile geri alınabilir. Sonrasında Navigation → Bake yapılmalıdır.
/// </summary>
public static class LevelBuilder
{
    private const string Layout = @"
################################
#........#......#....###########
#........#......q....###########
#..P.....#......q....###########
#....C...g......#....###########
#........g......#....#.........#
#......p.#.1.2.3#..N.#.....EEE.#
#........#......#....k.....EEE.#
#.########......#....k.....EEE.#
#.t....###.4.5.6#....#.....EEE.#
######.###......#....#.........#
#........#......#....###########
#........#.7.8.9#....###########
#...L....#......#....###########
#........#......#....###########
################################
";

    [MenuItem("Yazlab/Haritayı Oluştur")]
    public static void Build()
    {
        Tilemap ground = FindTilemap("Ground");
        Tilemap walls = FindTilemap("Walls");
        TileBase groundTile = FindTile("Ground");
        TileBase wallTile = FindTile("Wall");

        if (ground == null || walls == null || groundTile == null || wallTile == null)
        {
            EditorUtility.DisplayDialog("Harita oluşturulamadı",
                "Sahnede 'Ground' ve 'Walls' tilemap'leri, projede 'Ground' ve 'Wall' tile dosyaları olmalı.", "Tamam");
            return;
        }

        Undo.RegisterCompleteObjectUndo(new Object[] { ground, walls }, "Haritayı Oluştur");
        ground.ClearAllTiles();
        walls.ClearAllTiles();

        string[] rows = Layout.Replace("\r", "").Trim('\n').Split('\n');
        int height = rows.Length;
        int width = 0;
        var marks = new Dictionary<char, List<Vector3>>();

        for (int r = 0; r < height; r++)
        {
            string row = rows[r];
            width = Mathf.Max(width, row.Length);
            for (int x = 0; x < row.Length; x++)
            {
                char c = row[x];
                var cell = new Vector3Int(x, height - 1 - r, 0);

                if (c == '#') walls.SetTile(cell, wallTile);
                else if (c != ' ') ground.SetTile(cell, groundTile);

                if (c != '#' && c != '.' && c != ' ')
                {
                    if (!marks.TryGetValue(c, out var list)) marks[c] = list = new List<Vector3>();
                    list.Add(ground.GetCellCenterWorld(cell));
                }
            }
        }

        PlacePoint("Player", marks, 'P', 1.5f);
        PlacePoint("Companion", marks, 'C', 0.5f);
        PlacePoint("Lever", marks, 'L', 0.5f);
        PlacePoint("Guard", marks, 'N', 1f);
        PlaceArea("GateDoor", marks, 'g');
        PlaceArea("GuardDoor", marks, 'k');
        PlaceArea("TunnelDoor", marks, 't');
        PlacePoint("PressurePlate", marks, 'p', 0.8f);
        PlaceArea("PuzzleDoor", marks, 'q');
        PlaceNumberPlates(marks);
        PlaceArea("EndZone", marks, 'E');

        Camera cam = Camera.main;
        if (cam != null)
        {
            Undo.RecordObject(cam.transform, "Kamera");
            Undo.RecordObject(cam, "Kamera");
            Vector3 center = ground.CellToWorld(new Vector3Int(0, 0, 0)) + new Vector3(width / 2f, height / 2f, 0f);
            cam.transform.position = new Vector3(center.x, center.y, -10f);
            cam.orthographicSize = Mathf.Max(height / 2f, width / 2f / (16f / 9f)) + 0.3f;
        }

        EditorSceneManager.MarkSceneDirty(EditorSceneManager.GetActiveScene());
        Debug.Log("[LevelBuilder] Harita oluşturuldu. Şimdi 'Navigation' objesinde Bake yapın ve sahneyi kaydedin (Ctrl+S).");
    }

    private static void PlacePoint(string objectName, Dictionary<char, List<Vector3>> marks, char key, float scale)
    {
        GameObject go = GameObject.Find(objectName);
        if (go == null || !marks.TryGetValue(key, out var points))
        {
            Debug.LogWarning($"[LevelBuilder] '{objectName}' bulunamadı, atlandı.");
            return;
        }
        Undo.RecordObject(go.transform, "Yerleştir");
        go.transform.SetPositionAndRotation(points[0], Quaternion.identity);
        go.transform.localScale = new Vector3(scale, scale, 1f);
    }

    private static void PlaceArea(string objectName, Dictionary<char, List<Vector3>> marks, char key)
    {
        GameObject go = GameObject.Find(objectName);
        if (go == null || !marks.TryGetValue(key, out var points))
        {
            Debug.LogWarning($"[LevelBuilder] '{objectName}' bulunamadı, atlandı.");
            return;
        }

        Vector3 min = points[0], max = points[0];
        foreach (Vector3 p in points) { min = Vector3.Min(min, p); max = Vector3.Max(max, p); }

        Undo.RecordObject(go.transform, "Yerleştir");
        go.transform.SetPositionAndRotation((min + max) / 2f, Quaternion.identity);
        go.transform.localScale = new Vector3(max.x - min.x + 1f, max.y - min.y + 1f, 1f);
    }

    /// <summary>1-9 işaretli hücrelere numaralı plakaları koyar; yoksa oluşturur. Bulmaca objesini de ekler.</summary>
    private static void PlaceNumberPlates(Dictionary<char, List<Vector3>> marks)
    {
        Sprite sprite = AssetDatabase.LoadAssetAtPath<Sprite>("Assets/Art/Square.png");
        if (sprite == null)
            foreach (string guid in AssetDatabase.FindAssets("Square t:Sprite"))
                sprite = sprite != null ? sprite : AssetDatabase.LoadAssetAtPath<Sprite>(AssetDatabase.GUIDToAssetPath(guid));

        for (char d = '1'; d <= '9'; d++)
        {
            if (!marks.TryGetValue(d, out var points)) continue;

            string objectName = "Plate" + d;
            GameObject go = GameObject.Find(objectName);
            if (go == null)
            {
                go = new GameObject(objectName);
                Undo.RegisterCreatedObjectUndo(go, "Plaka oluştur");
                var sr = go.AddComponent<SpriteRenderer>();
                sr.sprite = sprite;
                sr.sortingOrder = 1;
                var plate = go.AddComponent<NumberPlate>();
                var so = new SerializedObject(plate);
                so.FindProperty("number").intValue = d - '0';
                so.ApplyModifiedPropertiesWithoutUndo();
            }
            else
            {
                Undo.RecordObject(go.transform, "Yerleştir");
            }

            go.transform.SetPositionAndRotation(points[0], Quaternion.identity);
            go.transform.localScale = new Vector3(0.7f, 0.7f, 1f);
        }

        if (GameObject.Find("SequencePuzzle") == null && marks.ContainsKey('1'))
        {
            var puzzle = new GameObject("SequencePuzzle");
            Undo.RegisterCreatedObjectUndo(puzzle, "Bulmaca oluştur");
            puzzle.AddComponent<SequencePuzzle>();
        }
    }

    private static Tilemap FindTilemap(string name)
    {
        foreach (Tilemap t in Object.FindObjectsByType<Tilemap>(FindObjectsSortMode.None))
            if (t.name == name) return t;
        return null;
    }

    private static TileBase FindTile(string name)
    {
        foreach (string guid in AssetDatabase.FindAssets(name + " t:TileBase"))
        {
            var tile = AssetDatabase.LoadAssetAtPath<TileBase>(AssetDatabase.GUIDToAssetPath(guid));
            if (tile != null && tile.name == name) return tile;
        }
        return null;
    }
